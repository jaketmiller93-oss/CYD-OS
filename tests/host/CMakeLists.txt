cmake_minimum_required(VERSION 3.16)
project(pocket_pal_host C CXX ASM)
if(NOT EXISTS "${CYDOS_SOURCE}/components/app_runtime/runtime.c")
    message(FATAL_ERROR "Set CYDOS_SOURCE to a CYDOS 0.4 source checkout")
endif()
add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
set(LV_CONF_PATH "${CYDOS_SOURCE}/tests/ui_preview/lv_conf.h" CACHE STRING "")
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "")
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "")
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "")
add_subdirectory("${CYDOS_SOURCE}/managed_components/lvgl__lvgl" lvgl)
file(GLOB LUA_FILES "${CYDOS_SOURCE}/components/lua/vendor/*.c")
add_library(lua_host STATIC ${LUA_FILES})
target_include_directories(lua_host PUBLIC "${CYDOS_SOURCE}/components/lua/vendor")
target_compile_definitions(lua_host PRIVATE LUA_USE_C89 LUAI_MAXCCALLS=32)
set(CJSON "$ENV{IDF_PATH}/components/json/cJSON")
add_executable(pet_host host.c "${CYDOS_SOURCE}/components/app_services/services.c" "${CYDOS_SOURCE}/components/cyd_common/json.c" "${CYDOS_SOURCE}/components/cyd_common/zip.c" "${CJSON}/cJSON.c")
target_include_directories(pet_host PRIVATE "${CYDOS_SOURCE}/tests/platform_sdk" "${CYDOS_SOURCE}/components/app_runtime" "${CYDOS_SOURCE}/components/cyd_common/include" "${CYDOS_SOURCE}/components/app_services/include" "${CYDOS_SOURCE}/components/network/include" "${CJSON}")
target_compile_definitions(pet_host PRIVATE CYD_ROOT="/tmp/pocket-pal-host-data")
target_link_libraries(pet_host PRIVATE lua_host lvgl m)
target_link_options(pet_host PRIVATE -fsanitize=address,undefined)
add_executable(model_host model_host.c)
target_link_libraries(model_host PRIVATE lua_host m)
target_link_options(model_host PRIVATE -fsanitize=address,undefined)
