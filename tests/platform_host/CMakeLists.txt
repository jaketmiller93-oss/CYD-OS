cmake_minimum_required(VERSION 3.16)
project(cyd_platform_host C CXX ASM)
add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../ui_preview/lv_conf.h" CACHE STRING "")
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "")
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "")
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "")
add_subdirectory("../../managed_components/lvgl__lvgl" lvgl)
file(GLOB LUA_FILES "../../components/lua/vendor/*.c")
add_library(lua_host STATIC ${LUA_FILES})
target_include_directories(lua_host PUBLIC ../../components/lua/vendor)
target_compile_definitions(lua_host PRIVATE LUA_USE_C89 LUAI_MAXCCALLS=32)
set(CJSON "$ENV{IDF_PATH}/components/json/cJSON")
add_executable(platform_host runtime_host.c ../../components/app_services/services.c ../../components/cyd_common/json.c ../../components/cyd_common/zip.c "${CJSON}/cJSON.c")
target_include_directories(platform_host PRIVATE ../platform_sdk ../../components/cyd_common/include ../../components/app_services/include ../../components/network/include "${CJSON}")
target_link_libraries(platform_host PRIVATE lua_host lvgl m)

target_compile_definitions(platform_host PRIVATE CYD_ROOT="/tmp/cyd040-platform-resources")

target_link_options(platform_host PRIVATE -fsanitize=address,undefined)
